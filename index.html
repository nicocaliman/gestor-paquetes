Script SQL para Supabase (SQL Editor)

Ejecuta este código en el SQL Editor de Supabase para crear la tabla:

CREATE TABLE IF NOT EXISTS packages (
  id TEXT PRIMARY KEY,
  destinatario TEXT NOT NULL,
  localidadDestinatario TEXT NOT NULL,
  bultos TEXT DEFAULT '1',
  weight TEXT DEFAULT '0',
  expedidor TEXT NOT NULL,
  localidadExpedidor TEXT NOT NULL,
  price NUMERIC(10,2) DEFAULT 0,
  createdAt TIMESTAMPTZ DEFAULT NOW()
);
ALTER PUBLICATION supabase_realtime ADD TABLE packages;
ALTER TABLE packages ENABLE ROW LEVEL SECURITY;
CREATE POLICY "Public Access" ON packages FOR ALL USING (true) WITH CHECK (true);
